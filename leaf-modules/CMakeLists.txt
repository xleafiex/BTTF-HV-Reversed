# Modules live outside src so the game's recursive source glob never embeds them.
function(add_leaf_module target source)
    add_library(${target} SHARED ${source})
    add_dependencies(${target} reVC)
    get_target_property(engine_includes reVC INCLUDE_DIRECTORIES)
    get_target_property(engine_definitions reVC COMPILE_DEFINITIONS)
    target_include_directories(${target} PRIVATE ${engine_includes} "${PROJECT_SOURCE_DIR}/vendor/librw")
    target_compile_definitions(${target} PRIVATE ${engine_definitions} RW_GL3)
    target_link_libraries(${target} PRIVATE "${CMAKE_BINARY_DIR}/src/libreVC.dll.a" winmm)
    set_target_properties(${target} PROPERTIES PREFIX "" CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON
        LINK_DEPENDS "${CMAKE_BINARY_DIR}/src/libreVC.dll.a")
endfunction()
